CREATE DATABASE GestionDispositivosTI;
USE GestionDispositivosTI;
-- 1. PERSONA
CREATE TABLE Persona (
    id_persona INT IDENTITY(1,1) PRIMARY KEY,
    cedula VARCHAR(10) NOT NULL UNIQUE,
    nombres VARCHAR(100) NOT NULL,
    apellidos VARCHAR(100) NOT NULL,
    correo_institucional VARCHAR(150) NOT NULL UNIQUE,
    telefono VARCHAR(15) NULL
);
-- 2. USUARIO
CREATE TABLE Usuario (
    id_usuario INT IDENTITY(1,1) PRIMARY KEY,
    id_persona INT NOT NULL UNIQUE,
    contrasena_hash VARCHAR(255) NOT NULL,
    estado VARCHAR(15) NOT NULL DEFAULT 'Activo',
    fecha_creacion DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    ultimo_acceso DATETIME2 NULL,
    CONSTRAINT FK_Usuario_Persona FOREIGN KEY (id_persona) REFERENCES Persona(id_persona),
    CONSTRAINT CK_Usuario_Estado CHECK (estado IN ('Activo', 'Inactivo'))
);
-- 3. ROL
CREATE TABLE Rol (
    id_rol INT IDENTITY(1,1) PRIMARY KEY,
    nombre VARCHAR(30) NOT NULL UNIQUE,
    descripcion VARCHAR(150) NULL
);
-- 4. USUARIOROL
CREATE TABLE UsuarioRol (
    id_usuario INT NOT NULL,
    id_rol INT NOT NULL,
    fecha_asignacion DATETIME2 NOT NULL DEFAULT SYSDATETIME(),
    CONSTRAINT PK_UsuarioRol PRIMARY KEY (id_usuario, id_rol),
    CONSTRAINT FK_UsuarioRol_Usuario FOREIGN KEY (id_usuario) REFERENCES Usuario(id_usuario),
    CONSTRAINT FK_UsuarioRol_Rol FOREIGN KEY (id_rol) REFERENCES Rol(id_rol)
);

-- 5. ESTUDIANTE
CREATE TABLE Estudiante (
    id_estudiante INT IDENTITY(1,1) PRIMARY KEY,
    id_persona INT NOT NULL UNIQUE,
    codigo_estudiante CHAR(4) NOT NULL UNIQUE,
    carrera VARCHAR(100) NOT NULL,
    nivel INT NOT NULL,
    estado VARCHAR(15) NOT NULL DEFAULT 'Activo',
    CONSTRAINT FK_Estudiante_Persona FOREIGN KEY (id_persona) REFERENCES Persona(id_persona),
    CONSTRAINT CK_Estudiante_Estado CHECK (estado IN ('Activo', 'Inactivo')),
    CONSTRAINT CK_Estudiante_Nivel CHECK (nivel BETWEEN 0 AND 9)
);
-- 6. DOCENTE
CREATE TABLE Docente (
    id_docente INT IDENTITY(1,1) PRIMARY KEY,
    id_persona INT NOT NULL UNIQUE,
    codigo_docente VARCHAR(20) NOT NULL UNIQUE,
    estado VARCHAR(15) NOT NULL DEFAULT 'Activo',
    CONSTRAINT FK_Docente_Persona FOREIGN KEY (id_persona) REFERENCES Persona(id_persona),
    CONSTRAINT CK_Docente_Estado CHECK (estado IN ('Activo', 'Inactivo'))
);
